% 1: 1.5 min
\storypart{1 / 3\quad Classical encoding}
\begin{frame}{Correcting a bit error in memory}
\talktext
Error-correcting memory stores extra bits so a changed bit can be recovered.
\begin{center}
\begin{tikzpicture}[node distance=6mm,font=\small]
\node[draw=Navy,fill=Pale,align=center] (a) {message\\$\mathtt{0111}$};
\node[draw=Navy,fill=Pale,align=center,right=of a] (b) {encoder\\$\mathtt{01101001}$};
\node[draw=Navy,fill=Gold!15,align=center,right=of b] (c) {memory readout\\$\textcolor{Gold}{\mathtt{1}}\mathtt{1101001}$};
\node[draw=Navy,fill=Pale,align=center,right=of c] (d) {decoder\\$\mathtt{0111}$};
\draw[-{Stealth},thick] (a)--(b);
\draw[-{Stealth},thick] (b)--(c);
\draw[-{Stealth},thick] (c)--(d);
\end{tikzpicture}
\end{center}
The \emph{encoder} computes the stored bits from the message. The \emph{decoder} reconstructs the message from the values read back.

\textbf{Assumption:} at most one stored bit changes. The decoder does not know which position changed.
\takeaway{We will specify an eight-bit example and prove that its decoder corrects any one changed bit.}
\source{Memory application: IBM, Troubleshooting Memory issues. The eight-bit example is a teaching construction.}
\end{frame}

% 2: 2 min
\begin{frame}{From four message bits to eight stored values}
\talktext
XOR returns $1$ exactly when its two bits differ. Write it as $\oplus$.
\begin{center}
\begin{tabular}{c|cccc}
input pair & $00$ & $01$ & $10$ & $11$\\\hline
XOR & $0$ & $1$ & $1$ & $0$
\end{tabular}
\end{center}
The message is four bits $(b,a_1,a_2,a_3)$. A position is a three-bit address $u=(u_1,u_2,u_3)$.
At address $u$, start with $b$ and XOR each $u_i$ for which $a_i=1$.

For message $(0,1,1,1)$, the stored bit is the \emph{parity}, the XOR of the three address bits:
\[
p(u)=u_1\oplus u_2\oplus u_3,\qquad p(101)=1\oplus0\oplus1=0.
\]
\begin{center}
\begin{tabular}{c|cccccccc}
address $u$ & $000$&$001$&$010$&$011$&$100$&$101$&$110$&$111$\\\hline
stored $p(u)$ & $0$&$1$&$1$&$0$&$1$&$0$&$0$&$1$
\end{tabular}
\end{center}
\takeaway{The three address bits locate a stored value. The four message bits specify all eight stored values.}
\end{frame}

% 3: 1.5 min
\begin{frame}{Affine Boolean functions}
\talktext
A \emph{Boolean function} assigns a bit to every binary address, as in the preceding table. Write $\F_2=\{0,1\}$ with XOR addition.

The message coefficients define the \emph{affine Boolean function}
\[
f_{a,b}(u)=\underbrace{b}_{\text{constant}}\oplus
\underbrace{a_1u_1\oplus a_2u_2\oplus a_3u_3}_{\text{selected coordinates}},
\quad a=(a_1,a_2,a_3).
\]
A product $a_i u_i$ is $u_i$ when $a_i=1$, and $0$ when $a_i=0$.
\[
f_{100,0}(u)=u_1\ \longleftrightarrow\ \mathtt{00001111},
\qquad f_{000,1}(u)=1\ \longleftrightarrow\ \mathtt{11111111}.
\]
An allowed stored word is a \emph{codeword}. The \emph{code} consists of the $16$ evaluation words, one for each four-bit message.
\takeaway{The decoder knows this encoder, so it knows exactly which 16 stored words are allowed.}
\end{frame}

% 4: 2 min
\begin{frame}{From a truth table to a polynomial}
\talktext
The \emph{algebraic normal form} (ANF) is the unique XOR of products of distinct variables representing a Boolean function.
Products are AND: $u_1u_2=1$ only when both bits are $1$.
The \emph{degree} is the largest number of variables in a nonzero product.

Suppose the value at $000$ changes. Call the received table $r$.
The \emph{indicator} of $000$ is $1$ there and $0$ at all other addresses:
\[
\delta_{000}(u)=(1\oplus u_1)(1\oplus u_2)(1\oplus u_3).
\]
Each factor tests whether one address bit is zero. Their product tests all three.
\[
\begin{aligned}
r&=p\oplus\delta_{000}\\
 &=1\oplus u_1u_2\oplus u_1u_3\oplus u_2u_3\oplus u_1u_2u_3.
\end{aligned}
\]
\takeaway{Every allowed function has degree at most one. The received function has degree three, so its table is not an allowed codeword.}
\end{frame}

% 5: 2 min
\begin{frame}{Comparing neighboring table entries}
\talktext
\begin{columns}[c,onlytextwidth]
\column{.33\textwidth}\centering
\resizebox{.80\linewidth}{!}{\cubepicture{1}{1}{0}}
\par\scriptsize Below: address. Teal: $1$. White: $0$. Gold: first-bit edges. Ring: changed entry.
\column{.64\textwidth}\small
The direction $e_1=(1,0,0)$ toggles the first address bit. Vector XOR acts coordinate by coordinate.
\[
r(000)\oplus r(100)=1\oplus1=0.
\]
The \emph{Boolean derivative} records this comparison at each address:
\[
D_{e_i}f(u)=f(u\oplus e_i)\oplus f(u),
\]
where $e_i$ has a $1$ in position $i$ and zeros elsewhere.

Derivatives of $p$ and $r$ along $e_1$:
\[
D_{e_1}p=1,\quad D_{e_1}r=1\oplus(1\oplus u_2)(1\oplus u_3).
\]
\end{columns}
\takeaway{One changed vertex reverses exactly one of four disjoint edge comparisons in each fixed direction.}
\end{frame}

% 6: 2 min
\begin{frame}{Recovering the four message coefficients}
\talktext
How can we recover $a_i$? Toggling $u_i$ changes only the term $a_i u_i$:
\[
D_{e_i}f_{a,b}=a_i.
\]
The four disjoint $e_1$ edges therefore give four estimates of $a_1$.
\begin{center}
\begin{tabular}{c|cccc}
addresses & $000,100$ & $001,101$ & $010,110$ & $011,111$\\\hline
received XOR & \textcolor{Gold}{$0$} & $1$ & $1$ & $1$
\end{tabular}
\end{center}
At most one estimate is wrong, because the readout has at most one changed entry.
\begin{uncoverenv}<2->
Majority gives $a_1=1$. The other two directions give $a_2=a_3=1$.
\[
r(u)\oplus a_1u_1\oplus a_2u_2\oplus a_3u_3
\quad\longleftrightarrow\quad \mathtt{10000000}.
\]
\takeaway{The remaining seven zeros recover $b=0$. The decoded message is $(0,1,1,1)$.}
\end{uncoverenv}
\end{frame}

% 7: 2.5 min
\begin{frame}{Why distinct codewords differ in at least four positions}
\talktext
\setlength{\abovedisplayskip}{2pt}\setlength{\belowdisplayskip}{1pt}
The \emph{weight} $\wt(f)$ counts the ones in the evaluation word.
The \emph{Hamming distance} $d(f,g)$ counts positions where two words differ.
\begin{columns}[c,onlytextwidth]
\column{.28\textwidth}\centering
\resizebox{.69\linewidth}{!}{\cubepicture{0}{1}{0}}
\par\scriptsize Below: $u$. Circle: $p(u)$. Teal $1$, white $0$. Gold pairs.
\column{.69\textwidth}
Each pair has one zero and one one, so $\wt(p)=4$.
For any nonconstant affine function, pair across a coordinate with coefficient $1$.
\[
H=\{u:p(u)=1\}=\{100,010,001,111\}.
\]
Such a level set is called an \emph{affine hyperplane}.
\end{columns}
\begin{uncoverenv}<2->
For distinct affine $f,g$, their XOR is either nonconstant or constant one:
\[
d(f,g)=\wt(f\oplus g)=4\ \text{or}\ 8,\quad\text{respectively}.
\]
Two codewords each one flip from $r$ would differ in at most two places. Distance at least four makes one-error recovery unique.
\end{uncoverenv}
\end{frame}

% 8: 2 min
\storypart{2 / 3\quad Quantum construction}
\begin{frame}{Counting intersections of Boolean supports}
\talktext
The \emph{support} $\supp(f)$ is the set of addresses where $f$ has value $1$.
For functions $f,g$, the product $fg$ selects their common support.
\begin{columns}[c,onlytextwidth]
\column{.40\textwidth}\centering\resizebox{.80\linewidth}{!}{\facepicture}
\par\scriptsize Dots: addresses. Teal: $u_1=1$. Gold: $u_2=1$. Dark edge: both equal $1$.
\column{.57\textwidth}
\begin{tabular}{ccl}
function & weight & support\\\hline
$u_1$ & $4$ & teal face\\
$u_1u_2$ & $2$ & dark common edge\\
$u_1u_2u_3$ & $1$ & endpoint $111$
\end{tabular}
\end{columns}
Ordinary addition counts a common $1$ twice. XOR counts it zero times:
\[
\wt(f\oplus g)=\wt(f)+\wt(g)-2\wt(fg).
\]
\end{frame}

% 9: 2 min
\begin{frame}{Single-qubit rotations produce phases}
\talktext
A qubit is a unit vector in $\mathbb C^2$. Its computational basis vectors are
$\ket0=(1,0)^{\mathsf T}$ and $\ket1=(0,1)^{\mathsf T}$.
A \emph{relative phase} multiplies one component by $e^{i\theta}$ for a real angle $\theta$:
\[
\frac{\ket0+\ket1}{\sqrt2}\ \longmapsto\
\frac{\ket0+e^{i\theta}\ket1}{\sqrt2}.
\]
For $\theta=\pi/4$, write $\omega=e^{i\pi/4}$ and call the rotation $T$.
Its inverse $T^\dagger$ uses $\omega^{-1}$.
For an $n$-bit physical word $v$, $\ket v$ is its basis vector.
Applying $T$ separately at all $n$ sites, written $T^{\otimes n}$, multiplies one factor $\omega$ for each $1$:
\[
T=\operatorname{diag}(1,\omega),\qquad
T^{\otimes n}\ket v=\omega^{\wt(v)}\ket v.
\]

On three qubits we want only the basis vector $\ket{111}$ to change sign.
This gate is \emph{controlled-controlled-$Z$} (CCZ):
\[
\CCZ\ket{a_1a_2a_3}=(-1)^{a_1a_2a_3}\ket{a_1a_2a_3}.
\]
\end{frame}

% 10: 2 min
\begin{frame}{Encoding three qubits with complementary words}
\talktext
Keep the three-bit label $a$. Pair the words with $b=0$ and $b=1$:
\begin{center}
\begin{tabular}{c|cc}
label $a$ & $f_{a,0}$ & $f_{a,1}$\\\hline
$000$ & $\mathtt{00000000}$ & $\mathtt{11111111}$\\
$100$ & $\mathtt{00001111}$ & $\mathtt{11110000}$\\
$111$ & $\mathtt{01101001}$ & $\mathtt{10010110}$
\end{tabular}
\end{center}
The corresponding \emph{encoded basis state} is the normalized superposition
\[
\ket a_L=\frac{\ket{f_{a,0}}+\ket{f_{a,1}}}{\sqrt2}.
\]
Eight disjoint pairs give eight orthogonal states, encoding three logical qubits in eight physical qubits. A logical gate acts on these encoded states.
\takeaway{An operation that only multiplies physical basis vectors by phases preserves this encoding only if both words in each pair receive the same phase.}
\end{frame}

% 11: 1.5 min
\begin{frame}{Choosing $T$ and $T^\dagger$ with parity}
\talktext
\begin{columns}[c,onlytextwidth]
\column{.32\textwidth}\centering
\resizebox{.93\linewidth}{!}{\cubepicture{0}{0}{1}}
\par\scriptsize Below: physical address $u$. White: $p(u)=0$, apply $T$. Teal: $p(u)=1$, apply $T^\dagger$.
\column{.65\textwidth}\small
The word $v=p$ has its four ones at $T^\dagger$ sites: its total rotation angle is $-4(\pi/4)$.

For any word $v$, count its ones at $T$ sites positively and its ones at $T^\dagger$ sites negatively. This is its \emph{signed weight}:
\[
\sw_p(v)=\sum_{u\in\F_2^3}(-1)^{p(u)}v(u).
\]
The sum is an ordinary integer. The physical phase is $\omega^{\sw_p(v)}$.
\end{columns}
\takeaway{The gate pattern depends on the physical address. Its action on an encoded state still needs to be checked.}
\end{frame}

% 12: 2 min
\begin{frame}{Checking the eight-qubit logical gate}
\talktext
At one address, $p=0$ contributes $v$, while $p=1$ contributes $-v$.
Both cases give the integer identity
\[
(-1)^{p(u)}v(u)=(p\oplus v)(u)-p(u).
\]
Summing over addresses gives $\sw_p(v)=\wt(p\oplus v)-4$.
For $a\ne111$, $p\oplus f_{a,b}$ is nonconstant affine, hence has weight four.
\begin{center}
\begin{tabular}{c|cc|c}
logical label $a$ & $\sw_p(f_{a,0})$ & $\sw_p(f_{a,1})$ & phase\\\hline
$a\ne111$ & $0$ & $0$ & \uncover<2->{$+1$}\\
$a=111$ & $-4$ & $+4$ & \uncover<2->{$-1$}
\end{tabular}
\end{center}
\begin{uncoverenv}<2->
\takeaway{$\omega^{4}=\omega^{-4}=-1$. Both words in every pair receive the same phase, and exactly the logical label $111$ changes sign: CCZ.}
\end{uncoverenv}
\end{frame}

% 13: 2.5 min
\begin{frame}{What additional error protection requires}
\talktext
A bit error $X$ swaps $\ket0,\ket1$. A phase error $Z$ sends $\ket1$ to $-\ket1$ and fixes $\ket0$.
A \emph{stabilizer check} is an observable with value $+1$ on every encoded state.

Example: flipping all eight bits exchanges the two words in each pair, so $X^{\otimes8}$ is a check. A single-site $Z$ changes its outcome to $-1$.

The \emph{syndrome} lists check outcomes as bits: $0$ for $+1$, $1$ for $-1$.
For checks that flip selected sites, a single-site phase error has a column of responses:
\[
\text{columns }\binom10,\binom01,\binom11
\quad\Longrightarrow\quad
\text{first two errors give }\binom10\oplus\binom01=\binom11.
\]
A zero column hides one phase error. Equal columns cancel, hiding two. Nonzero, distinct columns exclude both cases.

\emph{Quantum distance} is the fewest sites on which a product of bit and phase errors can act undetected and change encoded information. The eight-qubit code has distance two. Distance three corrects an arbitrary one-qubit error.
\end{frame}

% 14: 2 min
\begin{frame}{Deleting a subspace leaves weights 24 and 32}
\talktext
\begin{columns}[c,onlytextwidth]
\column{.40\textwidth}
Six-bit addresses have parts $u\in\F_2^4$ and $w\in\F_2^2$.
Delete the subspace $w=00$, retaining
\[
\mathcal P=\{(u,w):w\ne0\}.
\]
\centering
\resizebox{\linewidth}{!}{\blockpicture{0}}
\par $64\text{ points}-16\text{ deleted}=48$.
\par\scriptsize Each dot is one address. Each block contains all 16 choices of $u$.
\column{.57\textwidth}
For coefficients $\alpha\in\F_2^4$, $\beta\in\F_2^2$, use binary dot products:
\[
\alpha\cdot u=\alpha_1u_1\oplus\cdots\oplus\alpha_4u_4
\]
and similarly for $\beta\cdot w$. Define
\[
\ell(u,w)=\alpha\cdot u\oplus\beta\cdot w.
\]
Every nonzero $\ell$ has $32$ ones on all $64$ sites. On the deleted subspace, $\ell=\alpha\cdot u$: eight ones if $\alpha\ne0$, zero otherwise.
On retained sites, written $\ell|_{\mathcal P}$,
\[
\wt(\ell|_{\mathcal P})=
\begin{cases}32-8=24,&\alpha\ne0,\\32-0=32,&\alpha=0,\ \beta\ne0.\end{cases}
\]
\end{columns}
\end{frame}

% 15: 2 min
\begin{frame}{Completing the 48-qubit code}
\talktext
The six address coordinates give words $S_1,\ldots,S_6$. Flipping their selected sites gives stabilizer checks. Their XOR combinations form a space $S$ of $64$ words.

Three logical functions give words $K_1,K_2,K_3$. Let $K(a)=a_1K_1\oplus a_2K_2\oplus a_3K_3$ for the logical label $a$.
\begin{columns}[c,onlytextwidth]
\column{.46\textwidth}
Add each stabilizer word to the logical word:
\[
\ket a_L=\frac1{8}\sum_{s\in S}\ket{K(a)\oplus s}.
\]
The sign function $\gamma$ selects $T$ at value $0$, $T^\dagger$ at value $1$.
\column{.51\textwidth}\centering
\resizebox{\linewidth}{!}{\blockpicture{1}}
\par\scriptsize Actual signs: gold $T$, teal $T^\dagger$. Rows $u_1u_2$, columns $u_3u_4$, ordered $00,01,11,10$.
\end{columns}
\takeaway{The site geometry, logical functions, and rotation signs have separate roles. All three specify the construction.}
\end{frame}

% 16: 3 min
\begin{frame}{Checking phases through row intersections}
\talktext
A \emph{generator row} is one of the nine evaluation words $K_1,K_2,K_3,S_1,\ldots,S_6$. For distinct rows $g,h,q$, products mean coordinatewise AND, hence intersect supports.
The integer expansion at each site is
\[
g\oplus h\oplus q=g+h+q-2(gh+gq+hq)+4ghq.
\]
For a word $v$, set $\sw_\gamma(v)=\sum_{j=1}^{48}(-1)^{\gamma_j}v_j$.
Congruence modulo $m$ means equality of integer remainders after division by $m$.
\[
\begin{aligned}
\sw_\gamma(g)&\equiv0\pmod8 &&\text{coefficient }1,\\
\uncover<2->{\sw_\gamma(gh)}&\uncover<2->{\equiv0\pmod4} &&\uncover<2->{\text{coefficient }-2,}\\
\uncover<3->{\sw_\gamma(ghq)}&\uncover<3->{\equiv\begin{cases}1\pmod2,&\{g,h,q\}=\{K_1,K_2,K_3\},\\0\pmod2,&\text{otherwise.}\end{cases}}
\end{aligned}
\]
\begin{uncoverenv}<3->
Four or more rows have coefficients divisible by eight. Only the logical triple contributes $4a_1a_2a_3$ modulo eight, producing CCZ.
\end{uncoverenv}
\end{frame}

\begin{frame}{Reed--Muller codes and finite-length bounds}
\talktext
Our classical code evaluates affine functions on three-bit addresses.
It is the \emph{Reed--Muller code} $\RM(1,3)$.
More generally, $\RM(d,m)$ evaluates degree-at-most-$d$ Boolean polynomials on all $m$-bit addresses.

A binary code closed under XOR is \emph{linear}. Its parameters $[n,k,d_{\min}]$ mean $n$ stored bits, $k$ message bits, and minimum distance $d_{\min}$. Our code is $[8,4,4]$.

Each codeword has $9$ possible readouts with at most one error: itself and its $8$ single-bit changes. Unique correction requires these sets to be disjoint.
\[
\underbrace{16}_{\text{codewords}}\,
\underbrace{(1+8)}_{\text{readouts per codeword}}
=144\le\underbrace{2^8}_{\text{all readouts}}=256.
\]
For $M$ length-$n$ codewords correcting one error, the same count gives $M(1+n)\le2^n$, the one-error \emph{Hamming bound}.
\takeaway{The bound limits the number of messages at a fixed length under a worst-case error guarantee.}
\end{frame}

\begin{frame}{Erasure recovery and channel capacity}
\talktext
An \emph{erasure} is a missing value with a known position, shown by $?$.
The same code gives two different examples:
\begin{center}
\begin{tabular}{cl}
readout & explanation\\\hline
$\mathtt{?1101001}$ & only parity's codeword fits the seven known values\\
$\mathtt{0??0?00?}$ & both $\mathtt{00000000}$ and $\mathtt{01101001}$ fit
\end{tabular}
\end{center}
Now erase each position independently with probability $\varepsilon$.
The \emph{rate} is message bits per stored bit, $R=k/n$, here $4/8=1/2$.
\emph{Capacity} is the supremum of rates allowing error probability to approach zero as length grows.
\begin{uncoverenv}<2->
For this binary erasure channel, $C=1-\varepsilon$. For example, at $\varepsilon=1/4$, rates below $3/4$ can be made reliable.

\emph{Optimal erasure decoding} finds the unique compatible codeword, or reports ambiguity.
For every target $R\in(0,1)$, a sequence of Reed--Muller codes has lengths growing, rates tending to $R$, and whole-word recovery probability tending to one when $\varepsilon<1-R$.
\source{Kudekar et al., Reed--Muller Codes Achieve Capacity on Erasure Channels, Theorem 27.}
\end{uncoverenv}
\end{frame}

\begin{frame}{Backup: specifying the two types of stabilizer checks}
\talktext
For length-$n$ binary words, define the binary inner product
$v\cdot z=\bigoplus_{j=1}^n v_jz_j$.
Let $C$ be the XOR span of all generator rows, and $S\subset C$ the span of the stabilizer rows.
The orthogonal space is $C^\perp=\{z:z\cdot c=0\text{ for every }c\in C\}$.
\[
\ket a_L=|S|^{-1/2}\sum_{s\in S}\ket{K(a)\oplus s}.
\]
Flips on a word $s\in S$ permute the summands. Phase errors on a word $z\in C^\perp$ multiply every summand by $+1$.
These give the $X$ and $Z$ checks of a \emph{Calderbank--Shor--Steane} (CSS) code.
\[
d_X=\min_{v\in C\setminus S}\wt(v),\qquad
d_Z=\min_{z\in S^\perp\setminus C^\perp}\wt(z).
\]
Here $d_X,d_Z$ are the minimum weights of undetected bit and phase errors that change the encoded information. Quantum distance is $\min(d_X,d_Z)$.
\end{frame}

\begin{frame}{Backup: the nine Campbell--Howard parity conditions}
\talktext
Let $K_1,K_2,K_3$ be logical rows and $S_1,\ldots,S_s$ independent stabilizer rows. Products intersect supports. Indices within a product are distinct.
The \emph{Campbell--Howard conditions} require these overlap parities:
\begin{center}
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{c|lll}
rows involved & individual & pair & triple\\\hline
logical only & $\wt(K_i)$ even & $\wt(K_iK_j)$ even & $\wt(K_1K_2K_3)$ odd\\
mixed & & $\wt(K_iS_l)$ even & $\wt(K_iK_jS_l)$ even\\
 & & & $\wt(K_iS_lS_t)$ even\\
stabilizer only & $\wt(S_l)$ even & $\wt(S_lS_t)$ even & $\wt(S_lS_tS_v)$ even
\end{tabular}
\end{center}
These nine parity conditions specify the class in the paper's length exclusion. They are weaker than the main talk's signed congruences modulo $8,4,2$.
\takeaway{A code satisfying the parity conditions still needs a separate check of whether single-qubit $T/T^\dagger$ rotations implement CCZ without additional logical correction.}
\end{frame}

\begin{frame}{Backup: affine equivalence changes input coordinates}
\talktext
For $m$-bit inputs, let $A$ be an invertible binary $m\times m$ matrix and $t\in\F_2^m$ a translation. Matrix operations here use XOR addition.
Functions $f,g$ are \emph{input-affine equivalent} if
\[
g(u)=f(Au\oplus t).
\]
The bijection of input addresses permutes table positions. It preserves weight and, for nonzero functions, degree. Applying the same permutation to two tables preserves their distance.
For a direction $v\in\F_2^m$, the derivatives satisfy
\[
D_vg(u)=D_{Av}f(Au\oplus t).
\]
\takeaway{Permuting physical sites preserves a code and its gate action when every generator word and rotation sign is transported. Translating syndrome entries changes the check data instead.}
\end{frame}

\begin{frame}{Backup: translating the location of an error}
\talktext
For $t=(1,0,0)$, let $\delta_t(u)$ be one exactly at address $t$.
Translate the received function by $t$:
\[
r_t(u)=r(u\oplus t)=p(u)\oplus1\oplus\delta_t(u).
\]
The error moves from $000$ to $100$. Parity becomes its complement because toggling the first input changes its output.
\[
\mathtt{11101001}\quad\longmapsto\quad\mathtt{10011110}.
\]
Every invertible affine substitution preserves the Reed--Muller family and all distances between its words.
\takeaway{The same decoding argument applies after a consistent relabeling of addresses.}
\end{frame}

\begin{frame}{Backup: input equivalence and output addition differ}
\talktext
\emph{Extended-affine equivalence} also permits adding an affine output function $\ell$:
\[
g(u)=f(Au\oplus t)\oplus\ell(u),\qquad \deg\ell\le1.
\]
Adding $\ell$ permutes the affine code, so distance to that code is preserved. Weight need not be preserved:
\[
f=0,\quad\ell=u_1\quad\Longrightarrow\quad
\wt(f)=0,\quad\wt(f\oplus\ell)=4\quad(m=3).
\]
For a Boolean indicator $c$ of allowed positions, deleting the origin leaves
\[
n=\wt(c)-c(0)
\]
positions. A translation can change $c(0)$, hence the number of retained positions.
\end{frame}

\begin{frame}{Backup: a translated hyperplane can change syndrome rank}
\talktext
In three-bit coordinates $(x,y,z)$, translate the face $z=1$ by $001$:
\[
\{001,011,101,111\}\ \longrightarrow\ \{000,010,100,110\}.
\]
Treat each point as the responses of three checks to a phase error at that site:
\[
\begin{pmatrix}0&0&1&1\\0&1&0&1\\1&1&1&1\end{pmatrix}
\ \longrightarrow\
\begin{pmatrix}0&0&1&1\\0&1&0&1\\0&0&0&0\end{pmatrix}.
\]
Both sets need two independent displacement directions, so their affine dimension is two. Their columns span binary vector spaces of dimensions three and two, their respective ranks.

A zero syndrome column can still have a nonzero logical label, hence a nonzero full generator column. This translation changes syndrome data and does not establish quantum-code equivalence.
\end{frame}

\begin{frame}{Backup: a polynomial factor confines the support}
\talktext
In the paper's rank-eight analysis, write a candidate indicator in variables $y_1,\ldots,y_8$. For polynomials $g,h$ in the first six variables, a vanishing component gives
\[
c=y_8(g\oplus y_7h)\quad\Longrightarrow\quad
\supp(c)\subseteq\{y:y_8=1\}.
\]
The factor $y_8$ forces $c=0$ on $y_8=0$.
For a containing hyperplane after a change of origin:
\begin{itemize}
\item Passing through zero bounds the actual column rank by seven.
\item Avoiding zero, with proper support, supplies a missing direction for projection. A sum of two supported points lies in the parallel hyperplane through zero.
\end{itemize}
\source{Application requires the paper's CSS, three-logical-qubit, $d_Z\ge3$, Campbell--Howard, and distinct-nonzero-syndrome hypotheses. In the rank-eight exclusion, $n\le38<128$ leaves a missing point in the containing hyperplane.}
\end{frame}

\begin{frame}{Backup: derivatives recover polynomial coefficients}
\talktext
For an index set $I\subseteq\{1,\ldots,m\}$, let $u_I=\prod_{i\in I}u_i$, with empty product $1$.
Write the algebraic normal form as
\[
f(u)=\bigoplus_{I\subseteq\{1,\ldots,m\}}c_Iu_I,\qquad c_I\in\F_2.
\]
Apply the derivatives in the directions indexed by $I$, then evaluate at zero:
\[
c_I=\left(\prod_{i\in I}D_{e_i}\right)f(0).
\]
The product of derivative operators means successive application. For our received table, $D_{e_1}D_{e_2}D_{e_3}r=1$.
Every derivative of order greater than the degree vanishes.
\takeaway{The third derivative on three variables XORs all eight entries. Every degree-at-most-two table passes this parity check.}
\end{frame}

\begin{frame}{Backup: low degree gives a minimum nonzero weight}
\talktext
For $u'=(u_1,\ldots,u_{m-1})$, write a nonzero function of degree at most $d$ as
\[
f(u',u_m)=g(u')\oplus u_mh(u').
\]
The two halves of its table are $g$ and $g\oplus h$, with $\deg g\le d$, $\deg h\le d-1$. Thus
\[
\wt(f)=\wt(g)+\wt(g\oplus h)\ge\wt(h).
\]
If $h\ne0$, induction gives $\wt(h)\ge2^{m-d}$. If $h=0$, the halves agree and the same bound follows by induction on $g$.
The boundary case $d=m$ only requires weight at least one.
\[
\RM(d,m):\quad\left[2^m,\ \sum_{j=0}^d\binom mj,\ 2^{m-d}\right],\quad 0\le d\le m.
\]
\source{The monomial $u_1\cdots u_d$ attains the bound. The dimension counts the available monomials.}
\end{frame}

\begin{frame}{Backup: all Fourier coefficients of the received table}
\talktext
For a three-bit coefficient vector $a$, use the parity $a\cdot u$.
The normalized correlation and its eight values are
\[
C_r(a)=\frac18\sum_{u\in\F_2^3}(-1)^{r(u)}(-1)^{a\cdot u},
\qquad C_r(111)=\frac34,\quad C_r(a)=-\frac14\ (a\ne111).

The unnormalized score $W_r(a)=8C_r(a)$ counts agreements minus disagreements.
Allowing a constant $b$ gives the distance formula
\[
d(r,f_{a,b})=\frac{8-(-1)^bW_r(a)}2.
\]
\takeaway{The largest absolute correlation finds a nearest affine word. Its sign selects the constant term. Here the closest word is parity, at distance one.}

\begin{frame}{Backup: integer expansion of a binary sum}
\talktext
For Boolean functions $f,g,h$, XOR inside a weight and ordinary integer arithmetic outside it give
\[
\begin{aligned}
\wt(f\oplus g\oplus h)={}&\wt(f)+\wt(g)+\wt(h)\\
&-2[\wt(fg)+\wt(fh)+\wt(gh)]+4\wt(fgh).
\end{aligned}
\]
The same formula holds for signed weight, since the identity holds at each coordinate before summation.
An intersection of $j$ distinct rows has integer coefficient $(-2)^{j-1}$.
\takeaway{Modulo eight, terms involving four or more rows vanish. Individual rows, pairs, and triples suffice to calculate the phase.}
\end{frame}

\begin{frame}{Backup: the eight-qubit quantum distance is two}
\talktext
Use the CSS description with $C=\RM(1,3)$ and $S=\{0,\mathbf1\}$, where $\mathbf1$ is the all-one word.
\[
d_X=\min_{v\in C\setminus S}\wt(v)=4.
\]
A phase error at one site changes the all-site flip check's outcome to $-1$, so its syndrome is nonzero.
Phase errors at sites $000$ and $100$ together have zero syndrome but overlap $f_{100,0}$ in one position. The pair changes encoded information:
\[
d_Z=\min_{z\in S^\perp\setminus C^\perp}\wt(z)=2,
\qquad d=\min(d_X,d_Z)=2.
\]
\takeaway{The classical affine distance is four. The quantum code also has phase errors to detect, and its distance is two.}
\end{frame}

\begin{frame}{Backup: checking every word in an encoded state}
\talktext
Let $a\in\F_2^3$ select logical rows and $y\in\F_2^s$ select stabilizer rows. Their XOR combination is
\[
c(a,y)=\bigoplus_{i=1}^3a_iK_i\ \oplus\ \bigoplus_{l=1}^s y_lS_l.

A sign pattern implements CCZ without extra logical correction exactly when
\[

Partitioning sites into blocks $B$, with $c_B,\gamma_B$ the restrictions to a block, gives the exact integer identity

\sw_\gamma(c)=\sum_B[\wt(\gamma_B\oplus c_B)-\wt(\gamma_B)].

\source{The stabilizer-containing overlap conditions remove dependence on $y$. The remaining eight logical inputs can then be checked at $y=0$.}
