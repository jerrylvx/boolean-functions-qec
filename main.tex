% 1 | 00:30--02:30
\storypart{1 / 4\quad Problem}
\begin{frame}[label=slide1]{Why a native CCZ gate}
\talktext
Fault-tolerant computation needs a non-Clifford logical gate.
CCZ changes the sign of $\ket{111}$ only, and Clifford gates plus CCZ are universal.
\par\medskip
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{Magic-state distillation}
Standard route. Prepares noisy resource states and distills them, at a large space-time overhead.
\end{block}
\column{.48\textwidth}
\begin{block}{Transversal gate}
A code whose physical phase rotations implement the logical gate directly on the data block.
\end{block}
\end{columns}
\par\medskip
\textbf{Target:} a CSS code with $k=3$ logical qubits and distance $d\ge3$ whose logical CCZ is
one layer of physical $T$ or $T^\dagger$, one gate per data qubit, with no ancilla and no correction.
We call such an implementation \emph{native}. A single-qubit gate cannot spread one fault to two qubits.
\takeaway{Question: how short can a native CCZ code with $d\ge3$ be?}
\source{Bravyi--Haah 2012 (distillation)\quad Campbell--Howard 2017 (CSS codes with transversal CCZ).}
\note{00:30--02:30. Prerequisite: Clifford hierarchy, magic states. Terms: native implementation. Narration: CCZ is diagonal and in the third Clifford level, so physical phase rotations can produce it. Distillation is the default and costs space and time. A transversal layer avoids that cost if a short enough code exists. Native means T or T-dagger on every data qubit and nothing else. Distance three is the smallest distance that corrects an arbitrary single-qubit error. Reveals: One state. Transition: Here is what we know about the shortest length.}
\end{frame}

% 3 | 04:00--06:30
\begin{frame}[label=slide3]{Warm-up: the eight-qubit cube code}
\talktext
\begin{columns}[c,onlytextwidth]
\column{.34\textwidth}\centering
\resizebox{!}{22mm}{\cubepicture{0}{0}{1}}
\par\scriptsize Qubits at addresses $u\in\F_2^3$. White: $p(u)=0$, apply $T$. Teal: $p(u)=1$, apply $T^\dagger$.
\column{.63\textwidth}
For $a\in\F_2^3$ let $f_a(u)=a_1u_1\oplus a_2u_2\oplus a_3u_3$.
\[
\ket a_L=\tfrac1{\sqrt2}\bigl(\ket{f_a}+\ket{f_a\oplus\mathtt{11111111}}\bigr).
\]
Pattern: $p(u)=u_1\oplus u_2\oplus u_3$. With $\omega=e^{i\pi/4}$, a word $v$ gets the phase $\omega^{\sw(v)}$,
\[
\sw(v)=\#\{\text{ones at }T\}-\#\{\text{ones at }T^\dagger\}=\w{p\oplus v}-4.
\]
\end{columns}
\begin{uncoverenv}<2->
Per site, $(p\oplus v)-p$ equals $v$ where $p=0$ and $-v$ where $p=1$.
Every nonconstant affine function of three bits has exactly four ones.
\begin{center}\renewcommand{\arraystretch}{1.05}
\begin{tabular}{l|c|c|c}
label & $p\oplus f_a$, $p\oplus f_a\oplus\mathtt1$ & $\sw$ & phase\\\hline
$a\ne111$ & nonconstant affine & $0,\ 0$ & $+1$\\
$a=111$ & $\mathtt{00000000}$, $\mathtt{11111111}$ & $-4,\ +4$ & $\omega^{\pm4}=-1$
\end{tabular}
\end{center}
\end{uncoverenv}
\note{04:00--06:30. Prerequisite: Basis states as binary words. Terms: f\_a affine function; p parity; sw signed weight; omega. Narration: Each logical basis state superposes a word and its complement. The rotation pattern is parity itself. Per site, the signed contribution is (p XOR v) minus p, so the sum is the distance from p minus four. For a different from 111, p XOR f\_a is a nonconstant affine function, weight four, so the phase is one. For a equal to 111, the two words are p and its complement, with signed weights minus four and plus four, both giving minus one. Both words in each pair get the same phase, so this is logical CCZ. Reveals: State 1: code and pattern. State 2: the weight fact and the phase table. Transition: This code works as a gate but not as a memory.}
\end{frame}

% 4 | 06:30--07:30
\begin{frame}[label=slide4]{The cube code has distance two}
\talktext
Write $Z_u$ for a phase flip on qubit $u$. On a word $v$, $Z_{000}Z_{100}$ gives the sign $(-1)^{v(000)\oplus v(100)}$. Both words of $\ket a_L$ give $a_1$, so
\[
Z_{000}Z_{100}\ket a_L=(-1)^{a_1}\ket a_L.
\]
The only $X$-type stabilizer is $X^{\otimes8}$. It commutes with this weight-2 error, so the error is undetected and acts as a logical $Z_1$.
\[
\qcode{8}{3}{2},\qquad d_Z=2,\quad d_X=4.
\]
\takeaway{Goal: keep the gate mechanism and raise $d_Z$ to $3$. That requires a larger code and a systematic criterion.}
\note{06:30--07:30. Prerequisite: The cube code. Terms: Z\_u; d\_Z, d\_X. Narration: A two-qubit Z error along a first-coordinate edge applies a logical phase that depends on a1. Its overlap with the all-ones stabilizer is two, which is even, so no X check sees it. Distance two detects single errors but cannot correct them. Reveals: One state. Transition: To search for longer codes, write any CSS code as a binary matrix.}
\end{frame}

% 5 | 07:30--09:30
\storypart{2 / 4\quad Boolean-function criterion}
\begin{frame}[label=slide5]{A CSS code as a binary matrix}
\talktext
\[
G=\begin{bmatrix}K\\S\end{bmatrix},\qquad
\mathcal C_2=\operatorname{rowspan}S\subset\mathcal C_1=\operatorname{rowspan}G.
\]
The $s$ rows of $S$ are $X$-stabilizer generators, the $3$ rows of $K$ are $X$-logical generators, and the $n$ columns are the physical qubits.
The logical basis state with label $x\in\F_2^3$ is a coset sum:
\[
\ket{x_L}=2^{-s/2}\sum_{y\in\F_2^s}\ket{K^{\!\top}x\oplus S^{\!\top}y}.
\]
Distances: $d_X=\min\{\w c:c\in\mathcal C_1\setminus\mathcal C_2\}$ and $d_Z=\min\{\w c:c\in\mathcal C_2^\perp\setminus\mathcal C_1^\perp\}$, with $\w c$ the Hamming weight.
\par\medskip
\textbf{Cube code:} $S=(\mathtt{11111111})$, and $K$ has rows $u_1=\mathtt{00001111}$, $u_2=\mathtt{00110011}$, $u_3=\mathtt{01010101}$.
\note{07:30--09:30. Prerequisite: CSS codes. Terms: G, K, S, C1, C2, coset sum, d\_X, d\_Z, Hamming weight. Narration: A CSS code is a nested pair of classical codes. G lists logical rows above stabilizer rows. Each logical basis state is a uniform sum over one coset of C2. The Z checks and Z logicals are determined by G and are not written. The cube code is the case with one stabilizer row and three coordinate functions as logical rows. Reveals: One state. Transition: A diagonal gate multiplies each word of the coset sum by a phase.}
\end{frame}

% 6 | 09:30--11:30
\begin{frame}[label=slide6]{The logical gate becomes a phase condition}
\talktext
A pattern $\Gamma\in\{\pm1\}^n$ applies $T$ where $\Gamma_j=+1$ and $T^\dagger$ where $\Gamma_j=-1$.
A basis word $c$ gets the phase $\omega^{\sw_\Gamma(c)}$, with
\[
\omega=e^{i\pi/4},\qquad \sw_\Gamma(c)=\sum_j\Gamma_jc_j\quad\text{(signed weight)}.
\]
\textbf{Native CCZ condition:} for every logical label $x$ and stabilizer label $y$,
\[
\sw_\Gamma\bigl(K^{\!\top}x\oplus S^{\!\top}y\bigr)\equiv4x_1x_2x_3\pmod8.
\]
\begin{center}\renewcommand{\arraystretch}{1.25}
\begin{tabular}{ll}
\toprule
requirement & reason\\
\midrule
phase independent of $y$ & all terms of one coset sum must keep their relative phase\\
phase equal to $(-1)^{x_1x_2x_3}$ & the logical action must be CCZ\\
\bottomrule
\end{tabular}
\end{center}
\note{09:30--11:30. Prerequisite: The coset sum. Terms: Gamma; signed weight sw; omega. Narration: The phase of a word is omega to the signed weight. The exponent is defined mod eight. If two words of the same coset received different phases, the state would leave the code space. The target phase is four x1x2x3 in units of pi over four. Reveals: One state. Transition: The codewords are XORs of rows, so we need the signed weight of an XOR.}
\end{frame}

% 7 | 11:30--13:30
\begin{frame}[label=slide7]{XOR turns the phase into overlap counts}
\talktext
For rows $a,b$, the coordinatewise product $a\wedge b$ marks common ones. At each coordinate, as integers,
\[
a\oplus b=a+b-2\,a\wedge b,\qquad
a\oplus b\oplus c=a+b+c-2(a\wedge b+a\wedge c+b\wedge c)+4\,a\wedge b\wedge c.
\]
\begin{uncoverenv}<2->
In general, an XOR of $m$ rows expands with coefficient $(-2)^{t-1}$ on each $t$-fold product.
The signed weight $\sw_\Gamma$ is a weighted sum of coordinates, so it respects these integer identities:
\[
\sw_\Gamma(a\oplus b\oplus c)=\textstyle\sum_{\text{rows}}\sw_\Gamma(a)-2\sum_{\text{pairs}}\sw_\Gamma(a\wedge b)+4\,\sw_\Gamma(a\wedge b\wedge c).
\]
Products of four or more rows have coefficients divisible by $8$ and drop out mod $8$.
\end{uncoverenv}
\begin{uncoverenv}<3->
\par\medskip\textbf{Check:} supports $\{1,2\},\{2,3\},\{2,4\}$ have weights $2$, pair overlaps $1$, triple overlap $1$:
$\ \w{a\oplus b\oplus c}=6-2(3)+4=4$.
\end{uncoverenv}
\note{11:30--13:30. Prerequisite: Phase condition. Terms: wedge as coordinatewise AND; overlap. Narration: XOR is not additive over the integers, so convert it. The pair identity says a common one is counted twice and removed twice. The coefficient of a t-fold product is minus two to the t minus one. Multiply each coordinate by Gamma\_j and sum: the same identity holds for signed weights. Mod eight, only rows, pairs, and triples survive. This is the Bravyi--Haah parity identity. Reveals: State 1: pair and triple identity. State 2: general coefficient and signed version. State 3: a numeric check. Transition: Match the surviving terms against four x1x2x3.}
\end{frame}

% 8 | 13:30--15:30
\begin{frame}[label=slide8]{Signed overlaps decide the native gate}
\talktext
A codeword $K^{\!\top}x\oplus S^{\!\top}y$ is the XOR of the rows $g$ selected by $(x,y)$.
Matching coefficients of $1$, $-2$, and $4$ against $4x_1x_2x_3$ gives:
\begin{block}{Native criterion (paper, Prop.~signed overlaps)}
$\Gamma$ implements CCZ natively if and only if, for distinct rows $g,g',g''$ of $G$,
\[
\begin{aligned}
\sw_\Gamma(g)&\equiv0\pmod8,\\
\sw_\Gamma(g\wedge g')&\equiv0\pmod4,\\
\sw_\Gamma(g\wedge g'\wedge g'')&\equiv
\begin{cases}1\pmod2,&\{g,g',g''\}=\{K_1,K_2,K_3\},\\0\pmod2,&\text{otherwise}.\end{cases}
\end{aligned}
\]
\end{block}
In the cube code, $\Gamma=(-1)^p$ meets all three, with the logical triple $u_1\wedge u_2\wedge u_3$ at the single site $111$.
\note{13:30--15:30. Prerequisite: The expansion mod 8. Terms: native criterion. Narration: A row term has coefficient one, so it must vanish mod eight. A pair term has coefficient minus two, so divisibility by four suffices. A triple term has coefficient four, so only its parity matters, and only the logical triple may be odd. The criterion checks a polynomial number of overlaps instead of all 2 to the 3 plus s codewords. Reveals: One state. Transition: Before choosing signs, ask what all-T gates plus a Clifford correction require.}
\end{frame}

% 9 | 15:30--17:30
\begin{frame}[label=slide9]{Quasi-transversal versus native}
\hypertarget<1>{anchor9}{}\talktext
\textbf{Quasi-transversal:} apply $T^{\otimes n}$, then a diagonal Clifford correction built from $P=\operatorname{diag}(1,i)$ and $CZ$ gates.
Campbell and Howard give nine parity conditions on $G$ that suffice. Hu, Liang, and Calderbank show they are also necessary:
\begin{center}\renewcommand{\arraystretch}{1.2}
\begin{tabular}{lll}
\toprule
logical rows & mixed & stabilizer rows\\
\midrule
K1: $\w{K_i}$ even & KS1: $\w{K_i\wedge S_l}$ even & S1: $\w{S_l}$ even\\
K2: $\w{K_i\wedge K_{i'}}$ even & KS2: $\w{K_i\wedge K_{i'}\wedge S_l}$ even & S2: $\w{S_l\wedge S_{l'}}$ even\\
K3: $\w{K_1\wedge K_2\wedge K_3}$ \textbf{odd} & KS3: $\w{K_i\wedge S_l\wedge S_{l'}}$ even & S3: $\w{S_l\wedge S_{l'}\wedge S_{l''}}$ even\\
\bottomrule
\end{tabular}
\end{center}
\textbf{Native} is the special case whose correction is $P^\dagger$ on single qubits, since $TP^\dagger=T^\dagger$.
So every native code meets the nine conditions. The converse fails (slide 19).
\vfill\hfill\hyperlink{backup-ch}{\beamerbutton{Index ranges?}}
\note{15:30--17:30. Prerequisite: Native criterion. Terms: quasi-transversal; CH conditions K1--K3, KS1--KS3, S1--S3. Narration: Campbell--Howard proved sufficiency, Hu--Liang--Calderbank necessity. The nine conditions are parities of the same row, pair, and triple overlaps, now unsigned. K3 is the only odd one. A native pattern is an all-T layer followed by P-dagger on the T-dagger sites, which is a diagonal Clifford, so native implies the conditions. P is the Clifford phase gate, renamed to avoid a clash with the stabilizer matrix S. The lower bound later excludes even the weaker quasi-transversal case. Reveals: One state. Backup trigger: exact index ranges. Transition: One Boolean function packages all these parities.}
\end{frame}

% 10 | 17:30--19:00
\begin{frame}[label=slide10]{One Boolean function packages every overlap}
\talktext
Let $G$ have $r$ rows and distinct columns. Its \emph{column indicator} $\chi_G\colon\F_2^r\to\F_2$ equals $1$ exactly on the columns of $G$.
For a test polynomial $m$ in the row variables $z_1,\ldots,z_r$,
\[
\langle\chi_G,m\rangle=\sum_{\text{columns }z}m(z)\pmod2.
\]
\begin{center}\renewcommand{\arraystretch}{1.05}
\begin{tabular}{ll}
\toprule
test polynomial $m$ & $\langle\chi_G,m\rangle$ is the parity of\\
\midrule
$z_i$ & the weight of row $i$\\
$z_iz_j$ & the overlap of rows $i$ and $j$\\
$z_iz_jz_l$ & the triple overlap of rows $i,j,l$\\
\bottomrule
\end{tabular}
\end{center}
The nine conditions prescribe $\langle\chi_G,m\rangle$ for every monomial of degree $1$ to $3$.
\takeaway{A code becomes a set of points in $\F_2^r$. Its overlap conditions become degree conditions on one Boolean function (Nezami--Haah).}
\note{17:30--19:00. Prerequisite: Overlap parities. Terms: column indicator chi\_G; inner product with a monomial. Narration: A column z contributes to the weight of row i exactly when z\_i is one. A pair overlap counts columns with z\_i z\_j equal to one. Distinct columns let the set of points replace the matrix. Nezami and Haah used this to classify small triorthogonal codes. We use it to construct. Reveals: One state. Transition: Choose the point set first.}
\end{frame}

% 12 | 20:00--21:30
\begin{frame}[label=slide12]{Distinct nonzero columns give $d_Z\ge3$}
\talktext
Column $j$ of $S$ is the syndrome $\beta_j\in\F_2^s$ of a $Z$ error on qubit $j$.
A $Z$ error $v$ of weight at most two has syndrome
\[
Sv=\beta_j\qquad\text{or}\qquad Sv=\beta_i\oplus\beta_j.
\]
If the columns are distinct and nonzero, both are nonzero: every such error is detected, whatever $K$ is.
\par\medskip
Such a column set is a set of points of the \emph{projective space} $\PG(s-1,2)$, the $2^s-1$ nonzero vectors of $\F_2^s$.
The code $\mathcal C_2$ is then called \emph{projective}.
\par\medskip
Cube code: $s=1$ and all eight columns equal $1$. The pair $\{000,100\}$ has syndrome $1\oplus1=0$, which is the distance-2 error of slide 4.
\takeaway{Projective $S$ gives $d_Z\ge3$ for every choice of logical rows.}
\note{20:00--21:30. Prerequisite: Syndromes of Z errors are sums of S columns. Terms: syndrome beta\_j; projective space PG(s-1,2); projective code. \begin{block}{Theorem (length lower bound)}

\end{block}
\begin{enumerate}\setlength{\itemsep}{1pt}

\end{enumerate}

\begin{frame}[label=backupFG48]{Why FG48 has only weights 24 and 32}
\hypertarget{backup-fg48}{}\talktext
\textbf{Question from slide 14:} why only two weights, and why is FG48 unique?
\par\medskip
The $48$ columns occupy $48$ of the $63$ points of $\PG(5,2)$, missing $15$.
Each hyperplane has $31$ points, so it meets the column set in at least $31-15=16$ and at most $31$ points.
8-divisibility forces the section size $48-\w{c_a}$ to be a multiple of $8$: $16$ or $24$, giving weights $32$ or $24$.

\[
N_{16}+N_{24}=63,\qquad16N_{16}+24N_{24}=48\cdot31\quad\Longrightarrow\quad N_{16}=3,\ N_{24}=60.
\]
Every projective 8-divisible $[48,6]$ code has this weight enumerator. The classification of projective 8-divisible codes of length at most 48 leaves only the solid complement.

\begin{frame}[label=backupAffine]{Affine changes of coordinates preserve everything}
